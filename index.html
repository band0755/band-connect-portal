function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.type === "handle_test") { var sheet = ss.getSheetByName("Handle_Testing"); var now = new Date(); var dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd"); var timeStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "HH:mm:ss"); sheet.appendRow([ dateStr, timeStr, data.serial || "", data.mac || "", data.led || "P", data.buzzer || "P", data.haptic1 || "P", data.haptic2 || "P", data.str || "P", "P", data.charge || "P", data.comments || "", data.c0lb || "P", data.c1lb || "P", data.c5lb || "P", data.c10lb || "P", data.strainReading || "" ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } else if (data.type === "kit_assembly") { var kitSheet = ss.getSheetByName("Kit_Assembly"); var now = new Date(); var dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd"); kitSheet.appendRow([ dateStr, data.kitSerial || "", "", data.leftSerial || "", data.rightSerial || "", data.carabinerReg ? true : false, data.carabinerSwiv ? true : false, data.phoneHolder ? true : false, data.doorAnchor ? true : false, data.cables ? true : false, data.block ? true : false, data.guide ? true : false, data.label ? true : false, data.refurb ? true : false ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var action = e.parameter.action; if (action === "getTestedHandles") { var ss = SpreadsheetApp.getActiveSpreadsheet(); var testSheet = ss.getSheetByName("Handle_Testing"); var rows = testSheet.getDataRange().getValues(); var handles = []; // Skip header row for (var i = 1; i < rows.length; i++) { var serial = rows[i][2]; // Col C (Index 2) is Serial No. var mac = rows[i][3]; // Col D (Index 3) is MAC ID var result = rows[i][9]; // Col J (Index 9) is Result if (serial && result === "P") { handles.push({ serial: String(serial), mac: String(mac) }); } } return ContentService.createTextOutput(JSON.stringify({handles: handles})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput("Band Connect Portal Active"); }
